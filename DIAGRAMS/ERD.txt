    %% ============================================================
    %% Pretty Nails Nail Salon Booking System - Entity Relationship Diagram
    %% ============================================================

    %% ============================================================
    %% ENTITIES
    %% ============================================================

    User {
        UUID id PK
        String email
        Boolean emailVerified
        String name
        String nickname
        String password
        String image
        String role
        String phone
        DateTime createdAt
        DateTime updatedAt
    }

    Account {
        UUID id PK
        UUID userId FK
        String type
        String provider
        String providerAccountId
        String refreshToken
        String accessToken
        DateTime expiresAt
        String tokenType
        String scope
        String idToken
        String sessionState
        DateTime createdAt
        DateTime updatedAt
    }

    Service {
        UUID id PK
        String name
        String category
        String description
        BigDecimal basePrice
        Integer duration
        Boolean isActive
        DateTime createdAt
        DateTime updatedAt
    }

    ServiceVariant {
        UUID id PK
        UUID serviceId FK
        String name
        BigDecimal price
        DateTime createdAt
        DateTime updatedAt
    }

    Booking {
        UUID id PK
        UUID userId FK
        String status
        Date appointmentDate
        Time startTime
        Time endTime
        BigDecimal totalPrice
        BigDecimal downpayment
        String notes
        String inspirationImg
        DateTime createdAt
        DateTime updatedAt
    }

    BookingService {
        UUID id PK
        UUID bookingId FK
        UUID serviceId FK
        UUID variantId FK
        Integer quantity
        BigDecimal price
        DateTime createdAt
        DateTime updatedAt
    }

    Payment {
        UUID id PK
        UUID bookingId FK,UK
        UUID userId FK
        BigDecimal amount
        String status
        String method
        String reference
        DateTime verifiedAt
        UUID verifiedBy FK
        DateTime createdAt
        DateTime updatedAt
    }

    Availability {
        UUID id PK
        Date date
        Boolean isAvailable
        Time startTime
        Time endTime
        Integer maxSlots
        String note
        DateTime createdAt
        DateTime updatedAt
    }

    LoyaltyCard {
        UUID id PK
        UUID userId FK,UK
        Integer stampCount
        DateTime lastStampAt
        DateTime expiresAt
        DateTime createdAt
        DateTime updatedAt
    }

    LoyaltyStamp {
        UUID id PK
        UUID cardId FK
        UUID bookingId FK
        DateTime earnedAt
        String source
        Integer amount
        DateTime createdAt
        DateTime updatedAt
    }

    CustomerNote {
        UUID id PK
        UUID userId FK
        UUID adminId FK
        String content
        String nickname
        DateTime createdAt
        DateTime updatedAt
    }

    Session {
        UUID id PK
        String sessionToken
        UUID userId FK
        DateTime expires
        DateTime createdAt
        DateTime updatedAt
    }

    VerificationToken {
        UUID id PK
        String identifier
        String token
        DateTime expires
        DateTime createdAt
    }

    Gallery {
        UUID id PK
        String title
        String description
        String imageUrl
        String category
        Boolean isActive
        DateTime createdAt
        DateTime updatedAt
    }

    Setting {
        UUID id PK
        String key
        String value
        String description
        DateTime createdAt
        DateTime updatedAt
    }

    %% ============================================================
    %% RELATIONSHIPS WITH CARDINALITIES
    %% ============================================================

    %% User - Account (One-to-Many: User 1 --- * Account)
    User ||--o{ Account : "accounts"

    %% User - Session (One-to-Many: User 1 --- * Session)
    User ||--o{ Session : "sessions"

    %% User - Booking (One-to-Many: User 1 --- * Booking)
    User ||--o{ Booking : "bookings"

    %% User - LoyaltyCard (One-to-ZeroOrOne: User 1 --- 0..1 LoyaltyCard) [unique]
    User ||--o| LoyaltyCard : "loyaltyCard"

    %% User - CustomerNote as customer (One-to-Many: User 1 --- * CustomerNote)
    User ||--o{ CustomerNote : "notes (customer)"

    %% User - CustomerNote as admin (One-to-Many: User 1 --- * CustomerNote)
    User ||--o{ CustomerNote : "adminNotes (admin)"

    %% User - Payment (One-to-Many: User 1 --- * Payment)
    User ||--o{ Payment : "payments"

    %% User - Payment as admin verifier (One-to-Many: User 1 --- * Payment)
    User ||--o{ Payment : "verifiedPayments"

    %% Service - ServiceVariant (One-to-Many: Service 1 --- * ServiceVariant)
    Service ||--o{ ServiceVariant : "variants"

    %% Service - BookingService (One-to-Many: Service 1 --- * BookingService)
    Service ||--o{ BookingService : "bookings"

    %% ServiceVariant - BookingService (One-to-Many: ServiceVariant 1 --- * BookingService)
    ServiceVariant ||--o{ BookingService : "bookingServices"

    %% Booking - BookingService (One-to-Many: Booking 1 --- * BookingService)
    Booking ||--o{ BookingService : "services"

    %% Booking - Payment (One-to-ZeroOrOne: Booking 1 --- 0..1 Payment) [unique]
    Booking ||--o| Payment : "payment"

    %% Booking - LoyaltyStamp (One-to-Many: Booking 1 --- * LoyaltyStamp)
    Booking ||--o{ LoyaltyStamp : "loyaltyStamps"

    %% LoyaltyCard - LoyaltyStamp (One-to-Many: LoyaltyCard 1 --- * LoyaltyStamp)
    LoyaltyCard ||--o{ LoyaltyStamp : "stamps"

    %% ============================================================
    %% LEGEND (as comments)
    %% ============================================================
    %% Cardinality Notation:
    %% ||--||  : One-to-One (Exactly one on both sides)
    %% ||--o{  : One-to-Many (One to zero or many)
    %% ||--o|  : One-to-ZeroOrOne (One to zero or one)
    %% }o--||  : Many-to-One (Zero or many to one)
    %% o|      : Zero-or-One (Optional 0..1)
    %% o{      : Zero-or-Many (Optional 0..*)
    %% |{      : One-or-Many (Mandatory 1..*)
    %% ||      : Exactly One (Mandatory 1..1)
    %% Key Types:
    %% PK = Primary Key
    %% FK = Foreign Key
    %% UK = Unique Constraint
