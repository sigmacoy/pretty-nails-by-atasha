    %% ============================
    %% USERS TABLE
    %% ============================
    users {
        UUID id PK
        VARCHAR(255) email UK "NN"
        TIMESTAMP email_verified
        VARCHAR(100) name
        VARCHAR(100) nickname
        VARCHAR(255) password
        VARCHAR(500) image
        VARCHAR(20) role "NN, DEFAULT 'CUSTOMER'"
        VARCHAR(20) phone
        TIMESTAMP created_at "NN"
        TIMESTAMP updated_at
    }

    %% ============================
    %% ACCOUNTS TABLE
    %% ============================
    accounts {
        UUID id PK
        UUID user_id FK "NN"
        VARCHAR(50) type "NN"
        VARCHAR(50) provider "NN"
        VARCHAR(255) provider_account_id "NN"
        TEXT refresh_token
        TEXT access_token
        INTEGER expires_at
        VARCHAR(50) token_type
        VARCHAR(255) scope
        TEXT id_token
        VARCHAR(255) session_state
        TIMESTAMP created_at "NN"
        TIMESTAMP updated_at
        %% UK(provider, provider_account_id)
    }

    %% ============================
    %% SERVICES TABLE
    %% ============================
    services {
        UUID id PK
        VARCHAR(100) name UK "NN"
        VARCHAR(20) category "NN"
        TEXT description
        INTEGER base_price "NN"
        INTEGER duration "NN"
        BOOLEAN is_active "NN, DEFAULT true"
        TIMESTAMP created_at "NN"
        TIMESTAMP updated_at
    }

    %% ============================
    %% SERVICE_VARIANTS TABLE
    %% ============================
    service_variants {
        UUID id PK
        UUID service_id FK "NN"
        VARCHAR(50) name "NN"
        INTEGER price "NN"
        TIMESTAMP created_at "NN"
        TIMESTAMP updated_at
    }

    %% ============================
    %% BOOKINGS TABLE
    %% ============================
    bookings {
        UUID id PK
        UUID user_id FK "NN"
        VARCHAR(20) status "NN, DEFAULT 'PENDING'"
        TIMESTAMP appointment_date "NN"
        VARCHAR(5) start_time "NN"
        VARCHAR(5) end_time "NN"
        INTEGER total_price "NN"
        INTEGER downpayment "NN, DEFAULT 10000"
        TEXT notes
        VARCHAR(500) inspiration_img
        TIMESTAMP created_at "NN"
        TIMESTAMP updated_at
    }

    %% ============================
    %% BOOKING_SERVICES TABLE
    %% ============================
    booking_services {
        UUID id PK
        UUID booking_id FK "NN"
        UUID service_id FK "NN"
        UUID variant_id FK
        INTEGER quantity "NN, DEFAULT 1"
        INTEGER price "NN"
        TIMESTAMP created_at "NN"
        TIMESTAMP updated_at
    }

    %% ============================
    %% PAYMENTS TABLE
    %% ============================
    payments {
        UUID id PK
        UUID booking_id UK, FK "NN"
        UUID user_id FK "NN"
        INTEGER amount "NN"
        VARCHAR(20) status "NN, DEFAULT 'PENDING'"
        VARCHAR(20) method "NN"
        VARCHAR(100) reference
        TIMESTAMP verified_at
        VARCHAR(255) verified_by
        TIMESTAMP created_at "NN"
        TIMESTAMP updated_at
    }

    %% ============================
    %% AVAILABILITIES TABLE
    %% ============================
    availabilities {
        UUID id PK
        TIMESTAMP date UK "NN"
        BOOLEAN is_available "NN, DEFAULT true"
        VARCHAR(5) start_time "NN, DEFAULT '08:00'"
        VARCHAR(5) end_time "NN, DEFAULT '22:00'"
        INTEGER max_slots "NN, DEFAULT 4"
        TEXT note
        TIMESTAMP created_at "NN"
        TIMESTAMP updated_at
    }

    %% ============================
    %% LOYALTY_CARDS TABLE
    %% ============================
    loyalty_cards {
        UUID id PK
        UUID user_id UK, FK "NN"
        INTEGER stamp_count "NN, DEFAULT 0"
        TIMESTAMP last_stamp_at
        TIMESTAMP expires_at "NN"
        TIMESTAMP created_at "NN"
        TIMESTAMP updated_at
    }

    %% ============================
    %% LOYALTY_STAMPS TABLE
    %% ============================
    loyalty_stamps {
        UUID id PK
        UUID card_id FK "NN"
        UUID booking_id FK
        TIMESTAMP earned_at "NN"
        VARCHAR(20) source "NN"
        INTEGER amount
        TIMESTAMP created_at "NN"
        TIMESTAMP updated_at
    }

    %% ============================
    %% CUSTOMER_NOTES TABLE
    %% ============================
    customer_notes {
        UUID id PK
        UUID user_id FK "NN"
        UUID admin_id FK "NN"
        TEXT content "NN"
        VARCHAR(100) nickname
        TIMESTAMP created_at "NN"
        TIMESTAMP updated_at
    }

    %% ============================
    %% SESSIONS TABLE
    %% ============================
    sessions {
        UUID id PK
        VARCHAR(255) session_token UK "NN"
        UUID user_id FK "NN"
        TIMESTAMP expires "NN"
        TIMESTAMP created_at "NN"
        TIMESTAMP updated_at
    }

    %% ============================
    %% VERIFICATION_TOKENS TABLE
    %% ============================
    verification_tokens {
        UUID id PK
        VARCHAR(255) identifier "NN"
        VARCHAR(255) token UK "NN"
        TIMESTAMP expires "NN"
        TIMESTAMP created_at "NN"
        %% UK(identifier, token)
    }

    %% ============================
    %% GALLERY TABLE
    %% ============================
    gallery {
        UUID id PK
        VARCHAR(200) title "NN"
        TEXT description
        VARCHAR(500) image_url "NN"
        VARCHAR(20) category
        BOOLEAN is_active "NN, DEFAULT true"
        TIMESTAMP created_at "NN"
        TIMESTAMP updated_at
    }

    %% ============================
    %% SETTINGS TABLE
    %% ============================
    settings {
        UUID id PK
        VARCHAR(100) key UK "NN"
        TEXT value "NN"
        TEXT description
        TIMESTAMP created_at "NN"
        TIMESTAMP updated_at
    }

    %% ============================
    %% RELATIONSHIPS
    %% ============================

    %% accounts -> users (One-to-Many)
    users ||--o{ accounts : "user_id"

    %% service_variants -> services (One-to-Many)
    services ||--o{ service_variants : "service_id"

    %% bookings -> users (One-to-Many)
    users ||--o{ bookings : "user_id"

    %% booking_services -> bookings (One-to-Many)
    bookings ||--o{ booking_services : "booking_id"

    %% booking_services -> services (One-to-Many)
    services ||--o{ booking_services : "service_id"

    %% booking_services -> service_variants (Many-to-Many via junction)
    service_variants ||--o{ booking_services : "variant_id"

    %% payments -> bookings (One-to-One)
    bookings ||--|| payments : "booking_id"

    %% payments -> users (One-to-Many)
    users ||--o{ payments : "user_id"

    %% loyalty_cards -> users (One-to-One)
    users ||--|| loyalty_cards : "user_id"

    %% loyalty_stamps -> loyalty_cards (One-to-Many)
    loyalty_cards ||--o{ loyalty_stamps : "card_id"

    %% loyalty_stamps -> bookings (Many-to-Many via junction)
    bookings ||--o{ loyalty_stamps : "booking_id"

    %% customer_notes -> users (customer) (One-to-Many)
    users ||--o{ customer_notes : "user_id (customer)"

    %% customer_notes -> users (admin) (One-to-Many)
    users ||--o{ customer_notes : "admin_id (admin)"

    %% sessions -> users (One-to-Many)
    users ||--o{ sessions : "user_id"


    %% **Legend:**
    %% - `PK` = Primary Key
    %% - `FK` = Foreign Key
    %% - `UK` = Unique Key
    %% - `NN` = Not Null
    %% - `||--o{` = One-to-Many
    %% - `||--||` = One-to-One
    %% - `}o--o{` = Many-to-Many (via junction table)
